cmake_minimum_required(VERSION 3.16)
project(path_follow_gazebo LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(path_follow_mpc REQUIRED)

add_library(path_follow_gazebo_drive SHARED src/drive_plugin.cpp)
target_compile_features(path_follow_gazebo_drive PRIVATE cxx_std_17)
target_compile_options(path_follow_gazebo_drive PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(path_follow_gazebo_drive SYSTEM PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(path_follow_gazebo_drive PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(path_follow_gazebo_drive ${GAZEBO_LIBRARIES})
ament_target_dependencies(path_follow_gazebo_drive gazebo_ros rclcpp geometry_msgs
  nav_msgs sensor_msgs tf2_ros ackermann_msgs path_follow_mpc)
install(TARGETS path_follow_gazebo_drive LIBRARY DESTINATION lib)
ament_package()
